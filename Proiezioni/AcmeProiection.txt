(P)_ACME ::=

/* --- RAMO 1: NOLEGGIO IMMEDIATO --- */
startRental@User ;
preAuthorize_OUT@Bank ;

(
    /* Caso Banca KO */
    insufficientFunds@Bank ;
    rentalFailed_OUT@User
    +
    /* Caso Banca OK */
    authToken@Bank ;
    unlock_OUT@Station ;

    (
        /* Caso Stazione Guasta */
        unlockFail@Station ;
        cancelAuth_OUT@Bank ;
        rentalFailed_OUT@User
        +
        /* Caso Stazione OK - Start Noleggio */
        unlockAck@Station ;
        startTracking_OUT@Fleet ;
        trackingAck@Fleet ;
        rentalStarted_OUT@User ;

        /* Loop Monitoraggio (ACME è attivo qui) */
        ( getStatus_OUT@Fleet ; statusReport@Fleet )* ;

        /* Chiusura Noleggio */
        endRental@User ;
        stopTracking_OUT@Fleet ;
        tripReport@Fleet ;
        lock_OUT@Station ;
        lockAck@Station ;

        /* Calcolo Costo */
        calculateCost_OUT@Calculator ;
        costReport@Calculator ;

        /* Pagamento Finale */
        commitPayment_OUT@Bank ;
        (
            paymentReceipt@Bank ;
            invoice_OUT@User
            +
            paymentFailed@Bank ;
            paymentSollecit_OUT@User
        )
    )
)

+

/* --- RAMO 2: PRENOTAZIONE --- */
reserve@User ;
preAuthorize_OUT@Bank ;

(
    /* Caso Banca KO */
    insufficientFunds@Bank ;
    reservationFailed_OUT@User
    +
    /* Caso Banca OK */
    authToken@Bank ;
    reserve_OUT@Station ;

    (
        /* Caso Veicolo non disponibile in Stazione */
        reserveFail@Station ;
        cancelAuth_OUT@Bank ;
        reservationFailed_OUT@User
        +
        /* Caso Stazione OK - Prenotazione Confermata */
        reserveAck@Station ;
        reservationConfirmed_OUT@User ;

        /* SCELTA TRIPLA (ACME riceve da User o da Timer) */
        (
            /* B3: Cancellazione Utente */
            cancelReservation@User ;
            cancelAuth_OUT@Bank ;
            cancelReservation_OUT@Station ;
            cancelReservationAck@Station ;
            (
                cancellationConfirmed_OUT@User
                +
                penaltyNotification_OUT@User
            )
            +
            /* B2: Timeout (Trigger ricevuto dal Timer) */
            timeoutExpired@Timer ;
            commitPenalty_OUT@Bank ;
            penaltyReceipt@Bank ;
            cancelReservation_OUT@Station ;
            cancelReservationAck@Station ;
            penaltyNotification_OUT@User
            +
            /* B1: Ritiro Veicolo */
            confirmPickup@User ;
            unlock_OUT@Station ;

            (
                /* Stazione KO */
                unlockFail@Station ;
                cancelAuth_OUT@Bank ;
                rentalFailed_OUT@User
                +
                /* Stazione OK - Flusso Standard */
                unlockAck@Station ;
                startTracking_OUT@Fleet ;
                trackingAck@Fleet ;
                rentalStarted_OUT@User ;

                ( getStatus_OUT@Fleet ; statusReport@Fleet )* ;

                endRental@User ;
                stopTracking_OUT@Fleet ;
                tripReport@Fleet ;
                lock_OUT@Station ;
                lockAck@Station ;

                calculateCost_OUT@Calculator ;
                costReport@Calculator ;

                commitPayment_OUT@Bank ;
                (
                    paymentReceipt@Bank ;
                    invoice_OUT@User
                    +
                    paymentFailed@Bank ;
                    paymentSollecit_OUT@User
                )
            )
        )
    )
)
