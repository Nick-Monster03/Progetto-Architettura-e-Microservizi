(P)_Station ::=

/* --- RAMO 1: NOLEGGIO IMMEDIATO --- */
unlock@ACME ;
(
    unlockFail_OUT@ACME
    +
    unlockAck_OUT@ACME ;
    lock@ACME ;
    lockAck_OUT@ACME
)

+

/* --- RAMO 2: PRENOTAZIONE --- */
reserve@ACME ;
(
    /* Veicolo non disponibile */
    reserveFail_OUT@ACME
    +
    /* Prenotazione accettata */
    reserveAck_OUT@ACME ;

    (
        /* B3 e B2: Cancellazione utente o Timeout — la Stazione riceve
           in entrambi i casi la stessa richiesta di rilascio del veicolo */
        cancelReservation@ACME ;
        cancelReservationAck_OUT@ACME
        +
        /* B1: Ritiro Veicolo */
        unlock@ACME ;
        (
            unlockFail_OUT@ACME
            +
            unlockAck_OUT@ACME ;
            lock@ACME ;
            lockAck_OUT@ACME
        )
    )
)
